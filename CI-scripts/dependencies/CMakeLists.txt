cmake_minimum_required(VERSION 3.26)

project(mhm_wheel_dependencies LANGUAGES C)

include(ExternalProject)

if(NOT CMAKE_BUILD_TYPE)
  set(CMAKE_BUILD_TYPE Release CACHE STRING "Build type" FORCE)
endif()

set(_common_args
  "-DCMAKE_BUILD_TYPE:STRING=${CMAKE_BUILD_TYPE}"
  "-DCMAKE_INSTALL_PREFIX:PATH=${CMAKE_INSTALL_PREFIX}"
  "-DCMAKE_POSITION_INDEPENDENT_CODE:BOOL=ON"
  "-DCMAKE_C_COMPILER:FILEPATH=${CMAKE_C_COMPILER}"
  "-DCMAKE_INSTALL_LIBDIR:STRING=lib"
)

foreach(_variable IN ITEMS
    CMAKE_OSX_ARCHITECTURES
    CMAKE_OSX_DEPLOYMENT_TARGET
    CMAKE_OSX_SYSROOT)
  if(DEFINED ${_variable} AND NOT "${${_variable}}" STREQUAL "")
    list(APPEND _common_args "-D${_variable}:STRING=${${_variable}}")
  endif()
endforeach()

set(_external_project_args
  DOWNLOAD_EXTRACT_TIMESTAMP FALSE
  TLS_VERIFY TRUE
  UPDATE_COMMAND ""
  TEST_COMMAND ""
)

if(MINGW)
  # Work around https://github.com/HDFGroup/hdf5/issues/5885.
  set(_hdf5_c_flags "-DCMAKE_C_FLAGS:STRING=-D_GNU_SOURCE")
endif()

ExternalProject_Add(zlib
  URL
    "https://github.com/madler/zlib/releases/download/v1.3.2/zlib-1.3.2.tar.gz"
  URL_HASH
    "SHA256=bb329a0a2cd0274d05519d61c667c062e06990d72e125ee2dfa8de64f0119d16"
  ${_external_project_args}
  CMAKE_ARGS
    ${_common_args}
    "-DZLIB_BUILD_SHARED:BOOL=ON"
    "-DZLIB_BUILD_STATIC:BOOL=OFF"
    "-DZLIB_BUILD_TESTING:BOOL=OFF"
    "-DZLIB_INSTALL:BOOL=ON"
)

ExternalProject_Add(hdf5
  DEPENDS zlib
  URL
    "https://github.com/HDFGroup/hdf5/releases/download/2.1.1/hdf5-2.1.1.tar.gz"
  URL_HASH
    "SHA256=efff93b5a904d66e8f626d7da60b5eedc9faf544be27dbabbaa87967b8ad798b"
  ${_external_project_args}
  CMAKE_ARGS
    ${_common_args}
    "-DCMAKE_PREFIX_PATH:PATH=${CMAKE_INSTALL_PREFIX}"
    "-DZLIB_ROOT:PATH=${CMAKE_INSTALL_PREFIX}"
    "-DBUILD_SHARED_LIBS:BOOL=ON"
    "-DBUILD_STATIC_LIBS:BOOL=OFF"
    "-DBUILD_TESTING:BOOL=OFF"
    "-DHDF5_ONLY_SHARED_LIBS:BOOL=ON"
    "-DHDF5_BUILD_HL_LIB:BOOL=ON"
    "-DHDF5_BUILD_TOOLS:BOOL=OFF"
    "-DHDF5_BUILD_UTILS:BOOL=OFF"
    "-DHDF5_BUILD_EXAMPLES:BOOL=OFF"
    "-DHDF5_BUILD_FORTRAN:BOOL=OFF"
    "-DHDF5_BUILD_CPP_LIB:BOOL=OFF"
    "-DHDF5_BUILD_JAVA:BOOL=OFF"
    "-DHDF5_BUILD_DOC:BOOL=OFF"
    "-DHDF5_ENABLE_ZLIB_SUPPORT:BOOL=ON"
    "-DHDF5_ENABLE_SZIP_SUPPORT:BOOL=OFF"
    "-DHDF5_ENABLE_PLUGIN_SUPPORT:BOOL=OFF"
    "-DHDF5_ENABLE_PARALLEL:BOOL=OFF"
    "-DHDF5_ENABLE_HDFS:BOOL=OFF"
    "-DHDF5_ENABLE_ROS3_VFD:BOOL=OFF"
    "-DHDF5_USE_ZLIB_STATIC:BOOL=OFF"
    ${_hdf5_c_flags}
)

ExternalProject_Add(netcdf
  DEPENDS hdf5
  URL
    "https://downloads.unidata.ucar.edu/netcdf-c/4.10.1/netcdf-c-4.10.1.tar.gz"
  URL_HASH
    "SHA256=db3b69ff4a5ee1a7d79a5c36664d2128b752c266e966369fcf7311ec5f927564"
  ${_external_project_args}
  CMAKE_ARGS
    ${_common_args}
    "-DCMAKE_PREFIX_PATH:PATH=${CMAKE_INSTALL_PREFIX}"
    "-DHDF5_ROOT:PATH=${CMAKE_INSTALL_PREFIX}"
    "-DZLIB_ROOT:PATH=${CMAKE_INSTALL_PREFIX}"
    "-DBUILD_SHARED_LIBS:BOOL=ON"
    "-DBUILD_TESTING:BOOL=OFF"
    "-DNETCDF_ENABLE_HDF5:BOOL=ON"
    "-DNETCDF_ENABLE_CDF5:BOOL=ON"
    "-DNETCDF_ENABLE_HDF4:BOOL=OFF"
    "-DNETCDF_ENABLE_REMOTE_FUNCTIONALITY:BOOL=OFF"
    "-DNETCDF_ENABLE_DAP:BOOL=OFF"
    "-DNETCDF_ENABLE_BYTERANGE:BOOL=OFF"
    "-DNETCDF_ENABLE_NCZARR:BOOL=OFF"
    "-DNETCDF_ENABLE_NCZARR_ZIP:BOOL=OFF"
    "-DNETCDF_ENABLE_S3:BOOL=OFF"
    "-DNETCDF_ENABLE_S3_AWS:BOOL=OFF"
    "-DNETCDF_ENABLE_S3_INTERNAL:BOOL=OFF"
    "-DNETCDF_ENABLE_PLUGINS:BOOL=OFF"
    "-DNETCDF_PLUGIN_INSTALL:BOOL=OFF"
    "-DNETCDF_ENABLE_FILTER_SZIP:BOOL=OFF"
    "-DNETCDF_ENABLE_FILTER_BZ2:BOOL=OFF"
    "-DNETCDF_ENABLE_FILTER_BLOSC:BOOL=OFF"
    "-DNETCDF_ENABLE_FILTER_ZSTD:BOOL=OFF"
    "-DNETCDF_ENABLE_LIBXML2:BOOL=OFF"
    "-DNETCDF_ENABLE_PNETCDF:BOOL=OFF"
    "-DNETCDF_ENABLE_PARALLEL4:BOOL=OFF"
    "-DNETCDF_BUILD_UTILITIES:BOOL=OFF"
    "-DNETCDF_ENABLE_TESTS:BOOL=OFF"
    "-DNETCDF_ENABLE_BENCHMARKS:BOOL=OFF"
)

add_custom_target(mhm-wheel-dependencies ALL DEPENDS netcdf)
